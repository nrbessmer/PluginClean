# Send2Trash moves files to the macOS Trash through Foundation's
# NSFileManager. That matters for two reasons: it does not trigger the
# Automation -> Finder permission prompt, and it is not fooled by quotes
# or backslashes in a path the way an AppleScript string is.
# The app falls back to Finder/AppleScript only for paths Send2Trash
# cannot touch without elevation (anything under /Library).
Send2Trash==2.1.0
